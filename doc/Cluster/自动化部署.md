## 1.拓扑发现与映射

### 1.1 使用 ibnetdiscover 进行拓扑发现

```
# 发现整个网络拓扑并输出原始数据
ibnetdiscover -p

# 生成便于阅读的文本格式拓扑
ibnetdiscover --output-format text > ib_topology.txt

# 生成带节点名称的拓扑（需配合opensm配置）
ibnetdiscover -s
```

#### 拓扑可视化处理

```
# 生成Graphviz格式的拓扑图
ibnetdiscover -g > ib_topology.dot

# 将dot文件转换为PNG图像
dot -Tpng ib_topology.dot -o ib_topology.png

# 生成PDF格式拓扑图
dot -Tpdf ib_topology.dot -o ib_topology.pdf
```

#### 拓扑解析脚本示例

```
#!/bin/bash
# ib_topology_parser.sh

# 提取交换机信息
ibnetdiscover | grep -A5 "SW" | grep -E "(Node|Port)" | head -20

# 提取主机端信息
ibnetdiscover | grep -A5 "CA" | grep -E "(Node|Port|System)" | head -30

# 生成简单的拓扑摘要
echo "=== IB网络拓扑摘要 ==="
echo "交换机数量: $(ibnetdiscover | grep -c \"SW\")"
echo "主机适配器数量: $(ibnetdiscover | grep -c \"CA\")"
echo "路由器数量: $(ibnetdiscover | grep -c \"RT\")"
```

### 1.2 拓扑信息解析字段说明

* **Node Type**: SW（交换机）、CA（通道适配器）、RT（路由器）
* **LID**: 本地标识符
* **GUID**: 全局唯一标识符
* **Port**: 物理端口信息
* **Link Speed**: 链路速率（如：EDR 100Gb/s、HDR 200Gb/s）

        - "net.core.wmem_max = 134217728"
        - "net.ipv4.tcp_rmem = 4096 87380 134217728"
        - "net.ipv4.tcp_wmem = 4096 65536 134217728"

      command: sysctl -p /etc/sysctl.d/99-ib.conf
  
    - name: 配置端口MTU

      loop: "{{ ib_ports }}"
  
    - name: 启用端口
      command: "ibportstate {{ item.port }} enable"
      loop: "{{ ib_ports }}"

    - name: 显示交换机信息

#### 3.2.2 性能监控

```
#!/usr/bin/env python3
# ufm_performance_monitor.py

import requests
import json
import time
from datetime import datetime

class UFMMonitor:
    def __init__(self, ufm_host, token):
        self.base_url = f"https://{ufm_host}/ufmRest"
        self.headers = {
            "Authorization": f"Bearer {token}",
            "Content-Type": "application/json"
        }
  
    def get_performance_metrics(self, switch_id=None):
        """获取性能指标"""
        endpoint = "/app/sx_metrics"
        if switch_id:
            endpoint += f"/switches/{switch_id}"
  
        response = requests.get(
            self.base_url + endpoint,
            headers=self.headers,
            verify=False
        )
        return response.json()
  
    def get_port_statistics(self, port_guid):
        """获取端口统计信息"""
        endpoint = f"/app/ports/{port_guid}/statistics"
        response = requests.get(
            self.base_url + endpoint,
            headers=self.headers,
            verify=False
        )
        return response.json()
  
    def get_alerts(self, severity=None):
        """获取告警信息"""
        endpoint = "/app/alerts"
        params = {}
        if severity:
            params['severity'] = severity
  
        response = requests.get(
            self.base_url + endpoint,
            headers=self.headers,
            params=params,
            verify=False
        )
        return response.json()
  
    def monitor_threshold(self, metric, threshold, interval=60):
        """阈值监控"""
        while True:
            data = self.get_performance_metrics()
            current_value = self.extract_metric(data, metric)
    
            if current_value > threshold:
                self.send_alert(f"{metric}超过阈值: {current_value}")
    
            time.sleep(interval)
```

# ufm_api_client.py
import requests

class UFMAPIClient:
    def __init__(self, host, username, password):
        self.host = host
        self.base_url = f"https://{host}/ufmRest/v1"
        self.session = self.authenticate(username, password)
  
    def authenticate(self, username, password):
        """认证并获取token"""
        auth_url = f"{self.base_url}/auth/token"
        response = requests.post(auth_url, auth=(username, password))
        response.raise_for_status()
        return response.json()['token']
  
    def get_topology(self):
        """获取网络拓扑"""
        url = f"{self.base_url}/topology"
        response = requests.get(url, headers=self.get_headers())
        return response.json()
  
    def get_performance_data(self, time_range="1h"):
        """获取性能数据"""
        url = f"{self.base_url}/performance/metrics"
        params = {'time_range': time_range}
        response = requests.get(url, headers=self.get_headers(), params=params)
        return response.json()
  
    def create_event(self, event_data):
        """创建事件"""
        url = f"{self.base_url}/events"
        response = requests.post(url, json=event_data, headers=self.get_headers())
        return response.json()
  
    def get_headers(self):
        return {
            "Authorization": f"Bearer {self.session}",
            "Content-Type": "application/json"
        }
```

#### 3.3.2 CLI工具封装

```
#!/bin/bash
# ufm_cli_wrapper.sh

UFM_HOST="ufm-server.local"
UFM_USER="admin"
UFM_PASSWORD="secure_password"

# 函数：UFM认证
ufm_authenticate() {
    TOKEN=$(curl -s -k -X POST \
        "https://${UFM_HOST}/ufmRest/v1/auth/token" \
        -H "Content-Type: application/json" \
        -d "{\"username\":\"${UFM_USER}\",\"password\":\"${UFM_PASSWORD}\"}" \
        | jq -r '.token')
  
    echo $TOKEN
}

# 函数：获取拓扑
ufm_get_topology() {
    TOKEN=$1
    curl -s -k -X GET \
        "https://${UFM_HOST}/ufmRest/v1/topology" \
        -H "Authorization: Bearer ${TOKEN}" \
        | jq .
}

# 函数：获取性能数据
ufm_get_performance() {
    TOKEN=$1
    METRIC=$2
    curl -s -k -X GET \
        "https://${UFM_HOST}/ufmRest/v1/performance/${METRIC}" \
        -H "Authorization: Bearer ${TOKEN}" \
        | jq .
}

# 函数：导出配置
ufm_export_config() {
    TOKEN=$1
    FORMAT=${2:-json}
    curl -s -k -X GET \
        "https://${UFM_HOST}/ufmRest/v1/config/export?format=${FORMAT}" \
        -H "Authorization: Bearer ${TOKEN}" \
        -o "ufm_config_$(date +%Y%m%d).${FORMAT}"
}

# 主函数
main() {
    ACTION=$1
    PARAM=$2
  
    TOKEN=$(ufm_authenticate)
  
    case $ACTION in
        "topology")
            ufm_get_topology $TOKEN
            ;;
        "performance")
            ufm_get_performance $TOKEN $PARAM
            ;;
        "export")
            ufm_export_config $TOKEN $PARAM
            ;;
        *)
            echo "用法: $0 {topology|performance|export}"
            ;;
    esac
}

main "$@"
```

##### UFM Prometheus Exporter

```
# prometheus_exporter.py
from prometheus_client import start_http_server, Gauge, Counter
import time
import requests

class UFMExporter:
    def __init__(self, ufm_client, port=9100):
        self.ufm = ufm_client
        self.port = port
  
        # 定义Prometheus指标
        self.ib_port_xmit_data = Gauge('ib_port_xmit_data_bytes',
                                      'IB端口发送数据量',
                                      ['switch', 'port'])
  
        self.ib_port_rcv_data = Gauge('ib_port_rcv_data_bytes',
                                     'IB端口接收数据量',
                                     ['switch', 'port'])
  
        self.ib_port_xmit_packets = Counter('ib_port_xmit_packets_total',
                                           'IB端口发送包数',
                                           ['switch', 'port'])
  
        self.ib_port_errors = Gauge('ib_port_errors_total',
                                   'IB端口错误计数',
                                   ['switch', 'port', 'error_type'])
  
        self.ib_link_speed = Gauge('ib_link_speed_gbps',
                                  'IB链路速率',
                                  ['switch', 'port', 'peer'])
  
        self.ib_temperature = Gauge('ib_switch_temperature_celsius',
                                   'IB交换机温度',
                                   ['switch'])
  
    def collect_metrics(self):
        """收集UFM指标"""
        topology = self.ufm.get_topology()
        performance = self.ufm.get_performance_data()
  
        # 处理端口统计
        for switch in topology['switches']:
            switch_name = switch['name']
    
            # 收集温度信息
            if 'temperature' in switch:
                self.ib_temperature.labels(switch=switch_name).set(
                    switch['temperature']
                )
    
            # 收集端口指标
            for port in switch['ports']:
                port_id = port['port_id']
                stats = self.ufm.get_port_statistics(port['guid'])
        
                self.ib_port_xmit_data.labels(
                    switch=switch_name,
                    port=port_id
                ).set(stats['xmit_data_bytes'])
        
                self.ib_port_rcv_data.labels(
                    switch=switch_name,
                    port=port_id
                ).set(stats['rcv_data_bytes'])
        
                self.ib_port_xmit_packets.labels(
                    switch=switch_name,
                    port=port_id
                ).inc(stats['xmit_packets'])
        
                # 错误计数
                for error_type, count in stats['errors'].items():
                    self.ib_port_errors.labels(
                        switch=switch_name,
                        port=port_id,
                        error_type=error_type
                    ).set(count)
        
                # 链路速率
                if 'link_speed' in port:
                    self.ib_link_speed.labels(
                        switch=switch_name,
                        port=port_id,
                        peer=port.get('peer_guid', 'unknown')
                    ).set(self.convert_speed(port['link_speed']))
  
    def run(self):
        """启动exporter"""
        start_http_server(self.port)
        print(f"Exporter运行在端口 {self.port}")
  
        while True:
            self.collect_metrics()
            time.sleep(30)  # 30秒收集一次
  
    @staticmethod
    def convert_speed(speed_str):
        """转换速率字符串为Gbps数值"""
        speed_map = {
            'SDR': 2.5,
            'DDR': 5.0,
            'QDR': 10.0,
            'FDR': 14.0,
            'EDR': 25.0,
            'HDR': 50.0,
            'NDR': 100.0
        }
        return speed_map.get(speed_str, 0.0)
```

# ufm_report_generator.py
import pandas as pd
from datetime import datetime, timedelta
import matplotlib.pyplot as plt

class UFMReportGenerator:
    def __init__(self, ufm_client):
        self.ufm = ufm_client
  
    def generate_daily_report(self):
        """生成日报"""
        end_time = datetime.now()
        start_time = end_time - timedelta(days=1)
  
        # 收集数据
        metrics = self.ufm.get_metrics_range(start_time, end_time)
        events = self.ufm.get_events_range(start_time, end_time)
        alerts = self.ufm.get_alerts_range(start_time, end_time)
  
        # 生成报告
        report = {
            'period': {'start': start_time, 'end': end_time},
            'summary': self.generate_summary(metrics),
            'topology_changes': self.get_topology_changes(),
            'performance_analysis': self.analyze_performance(metrics),
            'event_log': events,
            'alert_summary': self.summarize_alerts(alerts),
            'recommendations': self.generate_recommendations(metrics)
        }
  
        return report
  
    def generate_performance_plot(self, metric_data, output_file):
        """生成性能图表"""
        df = pd.DataFrame(metric_data)
  
        plt.figure(figsize=(12, 6))
        for port in df['port_id'].unique():
            port_data = df[df['port_id'] == port]
            plt.plot(port_data['timestamp'], port_data['throughput'],
                    label=f"Port {port}")
  
        plt.title('IB端口吞吐量趋势')
        plt.xlabel('时间')
        plt.ylabel('吞吐量 (Gbps)')
        plt.legend()
        plt.grid(True)
        plt.savefig(output_file)
        plt.close()
```

## 4. 监控与维护

### 4.1 健康检查脚本

```
#!/bin/bash
# ib_health_check.sh

# 颜色定义
RED='\033[0;31m'
GREEN='\033[0;32m'
YELLOW='\033[1;33m'
NC='\033[0m'

echo "=== InfiniBand网络健康检查 ==="
echo "时间: $(date)"
echo "==============================="

# 1. 检查IB子网管理器
echo -e "\n1. 检查子网管理器状态"
if systemctl is-active opensm >/dev/null 2>&1; then
    echo -e "${GREEN}✓ OpenSM运行正常${NC}"
else
    echo -e "${RED}✗ OpenSM未运行${NC}"
    systemctl status opensm --no-pager
fi

# 2. 检查IB设备
echo -e "\n2. 检查IB设备"
if ibv_devices >/dev/null 2>&1; then
    echo -e "${GREEN}✓ IB设备检测正常${NC}"
    ibv_devices
else
    echo -e "${RED}✗ 未检测到IB设备${NC}"
fi

# 3. 检查端口状态
echo -e "\n3. 检查IB端口状态"
ibstat_output=$(ibstat 2>/dev/null)
if [ $? -eq 0 ]; then
    echo -e "${GREEN}✓ IB端口状态正常${NC}"
    ibstat | grep -A3 "Port"
else
    echo -e "${RED}✗ 获取端口状态失败${NC}"
fi

# 4. 检查链路状态
echo -e "\n4. 检查IB链路状态"
if iblinkinfo >/dev/null 2>&1; then
    ACTIVE_LINKS=$(iblinkinfo | grep -c "ACTIVE")
    DOWN_LINKS=$(iblinkinfo | grep -c "DOWN")
    echo -e "活跃链路: ${GREEN}${ACTIVE_LINKS}${NC}"
    echo -e "断开链路: $([ $DOWN_LINKS -eq 0 ] && echo -e "${GREEN}0${NC}" || echo -e "${RED}${DOWN_LINKS}${NC}")"
else
    echo -e "${RED}✗ 获取链路信息失败${NC}"
fi

# 5. 检查性能计数器
echo -e "\n5. 检查性能计数器"
if perfquery >/dev/null 2>&1; then
    echo -e "${GREEN}✓ 性能计数器正常${NC}"
    # 检查错误计数器
    ERRORS=$(perfquery | grep -i error | wc -l)
    if [ $ERRORS -gt 0 ]; then
        echo -e "${YELLOW}⚠ 发现 ${ERRORS} 个性能错误${NC}"
    fi
else
    echo -e "${RED}✗ 查询性能计数器失败${NC}"
fi

# 6. 检查MTU设置
echo -e "\n6. 检查MTU设置"
WRONG_MTU=$(ibcheckerrors mtu 2>/dev/null | grep -c "ERROR")
if [ $WRONG_MTU -eq 0 ]; then
    echo -e "${GREEN}✓ MTU设置正常${NC}"
else
    echo -e "${RED}✗ 发现MTU设置错误${NC}"
    ibcheckerrors mtu
fi

# 7. 生成健康报告
echo -e "\n7. 生成健康检查报告"
REPORT_FILE="/tmp/ib_health_report_$(date +%Y%m%d_%H%M%S).txt"

{
    echo "IB网络健康检查报告"
    echo "生成时间: $(date)"
    echo "主机名: $(hostname)"
    echo "----------------------------------------"
    echo "1. OpenSM状态: $(systemctl is-active opensm 2>/dev/null || echo '未运行')"
    echo "2. IB设备数量: $(ibv_devices 2>/dev/null | wc -l)"
    echo "3. 活跃链路: ${ACTIVE_LINKS}"
    echo "4. 断开链路: ${DOWN_LINKS}"
    echo "5. MTU错误: ${WRONG_MTU}"
    echo "6. 性能错误: ${ERRORS}"
} > "$REPORT_FILE"

echo -e "报告已保存至: ${YELLOW}${REPORT_FILE}${NC}"

# 8. 发送告警（如果有问题）
if [ $DOWN_LINKS -gt 0 ] || [ $WRONG_MTU -gt 0 ] || [ $ERRORS -gt 0 ]; then
    echo -e "\n${YELLOW}警告: 发现网络问题，请及时处理${NC}"
    # 可以在这里添加发送告警邮件的代码
fi

echo -e "\n=== 检查完成 ==="
```
